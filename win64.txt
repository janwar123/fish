import matplotlib.pyplot as plt

############### Head/tails


import numpy as np

# Step 1: Total Games
N = 1000

# Step 2: Initialize Variables
total_flips = 0     #সব Game-এ মোট কতবার Toss হয়েছে
total_return = 0    #মোট কত টাকা পুরস্কার পেয়েছি

win = 0
lose = 0

# Step 3: Simulate Games
for game in range(N):

    heads = 0
    tails = 0
    flips = 0

    while abs(heads - tails) < 3:

        r = np.random.rand()

        if r < 0.5:
            heads = heads + 1
        else:
            tails = tails + 1

        flips = flips + 1

    # Step 4: Count Total Flips
    total_flips = total_flips + flips

    # Step 5: Calculate Win or Lose
    if flips < 8:
        total_return = total_return + 8
        win = win + 1

    elif flips > 8:
        lose = lose + 1

# Step 6: Calculate Cost and Profit
total_cost = total_flips
profit = total_return - total_cost

average_flips = total_flips / N

# Step 7: Display Result
print("Total Games =", N)
print("Total Flips =", total_flips)
print("Total Cost =", total_cost, "Rs.")
print("Total Return =", total_return, "Rs.")
print("Winning Games =", win)
print("Losing Games =", lose)
print("Average Flips =", round(average_flips, 2))
print("Total Profit/Loss =", profit, "Rs.")

import numpy as np

# Step 1: Given Information
N = 30000

life = [1000, 1100, 1200, 1300, 1400,
        1500, 1600, 1700, 1800, 1900]

prob = [0.10, 0.14, 0.24, 0.14, 0.12,
        0.10, 0.06, 0.05, 0.03, 0.02]

delay = [4, 6, 8]
delay_prob = [0.3, 0.6, 0.1]

bearing_cost = 20
downtime_cost = 5
repairman_cost = 25

np.random.seed(42)

# Step 2: Existing Policy
count1 = 0
total_delay1 = 0

for i in range(3):

    clock = 0

    while True:

        bearing_life = np.random.choice(life, p=prob)

        if clock + bearing_life > N:
            break

        clock = clock + bearing_life

        repair_delay = np.random.choice(delay, p=delay_prob)

        total_delay1 = total_delay1 + repair_delay

        count1 = count1 + 1

# Calculate Existing Policy Cost
replace_time1 = count1 * 20

downtime1 = total_delay1 + replace_time1

cost1 = count1 * bearing_cost
cost2 = downtime1 * downtime_cost
cost3 = replace_time1 * repairman_cost / 60

total_cost1 = cost1 + cost2 + cost3

print("EXISTING POLICY")
print("Bearing Replaced =", count1)
print("Total Delay =", total_delay1)
print("Total Downtime =", downtime1)
print("Total Cost =", round(total_cost1, 2))


# Step 3: Proposed Policy
count2 = 0
total_delay2 = 0

clock = 0

while True:

    b1 = np.random.choice(life, p=prob)
    b2 = np.random.choice(life, p=prob)
    b3 = np.random.choice(life, p=prob)

    first_failure = min(b1, b2, b3)

    if clock + first_failure > N:
        break

    clock = clock + first_failure

    repair_delay = np.random.choice(delay, p=delay_prob)

    total_delay2 = total_delay2 + repair_delay

    count2 = count2 + 1

# Calculate Proposed Policy Cost
total_bearings = count2 * 3

replace_time2 = count2 * 40

downtime2 = total_delay2 + replace_time2

cost4 = total_bearings * bearing_cost
cost5 = downtime2 * downtime_cost
cost6 = replace_time2 * repairman_cost / 60

total_cost2 = cost4 + cost5 + cost6

print("\nPROPOSED POLICY")
print("Replacement Events =", count2)
print("Bearing Replaced =", total_bearings)
print("Total Delay =", total_delay2)
print("Total Downtime =", downtime2)
print("Total Cost =", round(total_cost2, 2))


# Step 4: Compare Both Policies
print("\nCOST COMPARISON")

print("Existing Policy Cost =", round(total_cost1, 2))
print("Proposed Policy Cost =", round(total_cost2, 2))

saving = total_cost1 - total_cost2

print("Saving =", round(saving, 2))

if saving > 0:
    print("Proposed Policy is Better")
else:
    print("Existing Policy is Better")

################ Question dice:

import numpy as np
import matplotlib.pyplot as plt

# Set random seed
np.random.seed(42)

# Game settings
cost_per_trial = 1
target_difference = 20
prize = 100  # Assumed prize; change if necessary

# Monte Carlo simulation
def play_game():

    H = 0
    T = 0
    trials = 0

    while abs(H - T) < target_difference:

        # Roll a six-sided die
        dice = np.random.randint(1, 7)

        trials += 1

        # Odd = H, Even = T
        if dice % 2 == 1:
            H += 1
        else:
            T += 1

    # Find winner
    if H > T:
        winner = "H"
    else:
        winner = "T"

    # Calculate cost and net profit
    total_cost = trials * cost_per_trial
    net_profit = prize - total_cost

    return trials, H, T, winner, total_cost, net_profit


# Repeat the game 1000 times
games = 1000

results = []

for i in range(games):
    results.append(play_game())

# Convert results into NumPy array
results = np.array(results, dtype=object)

trials = results[:, 0].astype(int)
costs = results[:, 4].astype(float)
profits = results[:, 5].astype(float)

# Display results
print("GAMBLING GAME SIMULATION")
print("------------------------")
print("Total Games:", games)
print("Average Trials:", np.mean(trials))
print("Average Cost: $", np.mean(costs))
print("Average Net Profit/Loss: $", np.mean(profits))

print("H Wins:", np.sum(results[:, 3] == "H"))
print("T Wins:", np.sum(results[:, 3] == "T"))

print("Winning Probability of H:",
      np.mean(results[:, 3] == "H"))

# Visualization 1: Number of trials
plt.figure(figsize=(12, 5))

plt.subplot(1, 2, 1)
plt.hist(trials, bins=30, edgecolor="black")
plt.title("Distribution of Game Duration")
plt.xlabel("Number of Trials")
plt.ylabel("Frequency")

# Visualization 2: Profit and loss
plt.subplot(1, 2, 2)
plt.hist(profits, bins=30, edgecolor="black")
plt.axvline(0, color="red", linestyle="--",
            label="Break-even")
plt.title("Profit and Loss Distribution")
plt.xlabel("Net Profit / Loss ($)")
plt.ylabel("Frequency")
plt.legend()

plt.tight_layout()
plt.show()

############### QUestion prev:

import simpy
import numpy as np

# Input parameters
mean_interarrival = 5
mean_service = 4
max_customers = 1000

# Set random seed
np.random.seed(42)

# Create SimPy environment
env = simpy.Environment()

# Create a single server
server = simpy.Resource(env, capacity=1)

# Store simulation statistics
waiting_times = []
total_service_time = 0

# Customer process
def customer(env, customer_id, server):

    global total_service_time

    # Record customer arrival time
    arrival_time = env.now

    # Request the server (FIFO)
    with server.request() as request:

        yield request

        # Record service start time
        service_start = env.now

        # Calculate waiting time
        waiting_time = service_start - arrival_time
        waiting_times.append(waiting_time)

        # Generate service time
        service_time = np.random.exponential(
            mean_service
        )

        total_service_time += service_time

        # Serve customer
        yield env.timeout(service_time)


# Customer arrival process
def customer_generator(env, server):

    for i in range(max_customers):

        # Generate inter-arrival time
        interarrival_time = np.random.exponential(
            mean_interarrival
        )

        # Wait for next arrival
        yield env.timeout(interarrival_time)

        # Create new customer
        env.process(customer(env, i, server))


# Start customer generator
env.process(customer_generator(env, server))

# Run simulation
env.run()

# Simulation ending time
simulation_end_time = env.now

# Calculate average delay
average_delay = np.mean(waiting_times)

# Calculate average number in queue
average_queue = sum(waiting_times) / simulation_end_time

# Calculate server utilization
server_utilization = (
    total_service_time / simulation_end_time
) * 100

# Theoretical arrival and service rates
arrival_rate = 1 / mean_interarrival
service_rate = 1 / mean_service

print("M/M/1 QUEUE SIMULATION")
print("---------------------")

print("Total Customers:", max_customers)

print("Average Delay in Queue:",
      round(average_delay, 4), "minutes")

print("Average Number in Queue:",
      round(average_queue, 4))

print("Server Utilization:",
      round(server_utilization, 2), "%")

print("Simulation Ending Time:",
      round(simulation_end_time, 4), "minutes")

print("Arrival Rate:", arrival_rate)
print("Service Rate:", service_rate)
